// Get the map score
Integer[Text] GetMapsScore(Text[] _MapsToScore) {
    declare Integer[Text] MapScores;
    declare Integer Points = 0;
    foreach (MapUid in _MapsToScore) {
        // if it's the current map skip it
        if (MapUid == MapInfo.MapUid) continue;
        MapScores[MapUid] = Points;
        Points += 1;
    }
    return MapScores;
}

Integer[Text] GetMapsServerScore() {
    declare persistent Text[] Persistent_MapVote_ServerRecentlyPlayedMaps;
    return GetMapsScore(Persistent_MapVote_ServerRecentlyPlayedMaps);
}

Void GetMapsToVote(Boolean _UseLeastPlayedMaps) {
    declare persistent Integer[Text] Persistent_MapVote_MapsPlayCount;
    declare CMapInfo[] Maps;
    foreach (_MapInfo in MapList) {
        if (_MapInfo.MapUid != MapInfo.MapUid) Maps.add(_MapInfo);
    }
    if (_UseLeastPlayedMaps) {
        // Get the map scores
        // Higher score = more recent & more players have played that map recently
        // Get the maps with the lowest scores
        declare Integer[Text] MapScores = GetMapsServerScore();
        foreach (Player in Players) {
            foreach (MapUid => Points in GetMapsPlayerScore(Player)) {
                if (!MapScores.existskey(MapUid)) MapScores[MapUid] = Points;
                else MapScores[MapUid] += Points;
            }
        }

        // maps that aren't in the MapScores list yet haven't been played by any player or the server recently
        // assign them a score of 0
        foreach (Map in MapList) {
            if (!MapScores.existskey(Map.MapUid) && Map.MapUid != MapInfo.MapUid) MapScores[Map.MapUid] = 0;
        }

        // remove maps that don't exist on the server (map list changed...)
        foreach (MapUid => MapScore in MapScores) {
            if (GetMapIndex(MapUid) == -1) {
                MapScores.removekey(MapUid);
            }
        }

        // sort maps by score [0=>[a,b,c],1=>[d,e,f],2=>[g,h,i]]

        declare Text[][Integer] GroupedMaps;
        declare Integer CurrentGroup = 0;

        foreach (MapUid => Score in MapScores) {
            if (!GroupedMaps.existskey(Score)) GroupedMaps[Score] = [];
            GroupedMaps[Score].add(MapUid);
        }

        GroupedMaps = GroupedMaps.sortkey();
        log("Grouped map scores:");
        log(GroupedMaps);
        while (G_MapVote_MapsToVote.count < (ML::Clamp(S_Rows,1,2)*C_MapsPerRow)) {
            if (!GroupedMaps.existskey(CurrentGroup)) {
                CurrentGroup += 1;
                continue;
            }

            if (GroupedMaps[CurrentGroup].count == 0) {
                GroupedMaps.removekey(CurrentGroup);
            } else {
                declare MapIndex = ML::Rand(0,GroupedMaps[CurrentGroup].count-1);
                declare CMapInfo VoteMapInfo <=> GetMapInfo(GroupedMaps[CurrentGroup][MapIndex]);
                G_MapVote_MapsToVote.add(K_MapInfo {
                    MapUid = VoteMapInfo.MapUid,
                    Name = VoteMapInfo.Name,
                    Author = VoteMapInfo.AuthorNickName,
                    AuthorLogin = VoteMapInfo.AuthorLogin,
                    ThumbnailUrl = "file://Thumbnails/MapUid/"^VoteMapInfo.MapUid
                });
                GroupedMaps[CurrentGroup].removekey(MapIndex);
            }
            yield;
        }

        
    } else {
        for (I, 1, (ML::Clamp(S_Rows,1,2)*C_MapsPerRow)) {
            declare CMapInfo VoteMapInfo <=> Maps[ML::Rand(0,Maps.count-1)];
            G_MapVote_MapsToVote.add( K_MapInfo {
                MapUid = VoteMapInfo.MapUid,
                Name = VoteMapInfo.Name,
                Author = VoteMapInfo.AuthorNickName,
                AuthorLogin = VoteMapInfo.AuthorLogin,
                ThumbnailUrl = "file://Thumbnails/MapUid/"^VoteMapInfo.MapUid
            });
            Maps.remove(VoteMapInfo);
        }
    }
    declare Text Headers = """Accept: application/json
Content-Type: application/json
User-Agent: ServerPlugin;MapVote/{{{Version}}}""";
    declare Text MapUidsToRequest;
    declare persistent Text[Text] Persistent_MapVote_MapThumbnailUrls;
    foreach (Map in G_MapVote_MapsToVote) {
        MapUidsToRequest ^= Map.MapUid^",";
        if (Persistent_MapVote_MapThumbnailUrls.existskey(Map.MapUid) && Persistent_MapVote_MapThumbnailUrls[Map.MapUid] == "") Persistent_MapVote_MapThumbnailUrls.removekey(Map.MapUid);
        if(!Persistent_MapVote_MapThumbnailUrls.existskey(Map.MapUid)) {
            G_MapVote_MapInfoRequests[Map.MapUid] = (Http.CreateGet("https://trackmania.io/api/map/"^Map.MapUid, True));
        }
    }
}

Void StartVote() {
    declare netwrite K_MapInfo[] Net_MapVote_MapsToVote for Teams[0];
    declare netwrite Integer Net_MapVote_EndTime for Teams[0];

    G_MapVote_CanVote = True;

    declare persistent Text[] Persistent_MapVote_ServerRecentlyPlayedMaps;
    if (Persistent_MapVote_ServerRecentlyPlayedMaps.exists(MapInfo.MapUid)) {
        Persistent_MapVote_ServerRecentlyPlayedMaps.remove(MapInfo.MapUid);
    }
    Persistent_MapVote_ServerRecentlyPlayedMaps.add(MapInfo.MapUid);
    if (Persistent_MapVote_ServerRecentlyPlayedMaps.count > 15) {
        Persistent_MapVote_ServerRecentlyPlayedMaps.removekey(0);
    }
    
 
    //not enough maps
    if (MapList.count < (ML::Clamp(S_Rows,1,2)*C_MapsPerRow)+1) G_MapVote_CanVote = False;

    // jukeboxed map
    if (MapList.existskey(NextMapIndex) && MapList[NextMapIndex].MapUid != G_MapVote_NextMapUid) G_MapVote_CanVote = False;

    if (!G_MapVote_CanVote) return;


    G_MapVote_EndTime = Now+(S_VotingTime*1000)+1000;
    G_MapVote_MapsToVote = [];

    HoldMapUnloadRequest = True;
    GetMapsToVote(S_UseLeastPlayedMaps);
    UIManager.UIAll.ForceSpectator = True;

    Net_MapVote_MapsToVote = G_MapVote_MapsToVote;
    Net_MapVote_EndTime = G_MapVote_EndTime;
}

Void EndVote() {
    declare netwrite Integer[Text] Net_MapVote_VoteCounts for Teams[0];
    declare netwrite Text Net_MapVote_PickedMapUid for Teams[0];
    declare netwrite Integer Net_MapVote_SelectReason for Teams[0]; // 0 = random, 1 = majority vote, 2 = everyone, 3 = first
    declare Integer[Text] MapUidsCount;

    declare Integer NbVotes = G_MapVote_PlayersVotes.count;
    foreach (Vote in G_MapVote_PlayersVotes) {
        if (!MapUidsCount.existskey(Vote)) MapUidsCount[Vote] = 0;
        MapUidsCount[Vote] = MapUidsCount[Vote] + 1;
    }
    MapUidsCount = MapUidsCount.sortreverse();
    Net_MapVote_VoteCounts = MapUidsCount;
    declare Text[] SortedMaps;
    foreach (MapUid => VoteCount in MapUidsCount) {
        SortedMaps.add(MapUid);
    }
    Net_MapVote_SelectReason = 0;
    G_MapVote_PickedMap <=> GetMapInfo(G_MapVote_MapsToVote[ML::Rand(0,G_MapVote_MapsToVote.count-1)].MapUid);
    if (NbVotes >= 1) {
        Net_MapVote_SelectReason = 1;
        // if vote counts are a tie pick the map that was voted for first
        if (SortedMaps.count > 1 && MapUidsCount[SortedMaps[0]] == MapUidsCount[SortedMaps[1]]) {
            Net_MapVote_SelectReason = 3;
            declare Text[] TiedMapUids;
            declare Integer VoteCount = MapUidsCount[SortedMaps[0]];
            foreach (MapUid => Count in MapUidsCount) {
                if (Count == VoteCount) TiedMapUids.add(MapUid);
            }

            declare Integer[Text] MaxVoteTimes;
            foreach (Login => Vote in G_MapVote_PlayersVotes) {
                if (TiedMapUids.exists(Vote)) {
                    if (!MaxVoteTimes.existskey(Vote)) MaxVoteTimes[Vote] = G_MapVote_VoteTimes[Login];
                    else {
                        if (G_MapVote_VoteTimes[Login] > MaxVoteTimes[Vote]) MaxVoteTimes[Vote] = G_MapVote_VoteTimes[Login];
                    }
                }
            }
            // find map with the earliest last vote
            MaxVoteTimes = MaxVoteTimes.sort();
            declare Text MinVoteTimeMap;
            foreach (MapUid => Time in MaxVoteTimes) {
                MinVoteTimeMap = MapUid;
                break;
            }
            G_MapVote_PickedMap <=> GetMapInfo(MinVoteTimeMap);
        } else {
            if (NbVotes == MapUidsCount[SortedMaps[0]]) {
                Net_MapVote_SelectReason = 2;
            }
            G_MapVote_PickedMap <=> GetMapInfo(SortedMaps[0]);
        }
    }
    Net_MapVote_PickedMapUid = G_MapVote_PickedMap.MapUid;
    NextMapIndex = MapList.keyof(G_MapVote_PickedMap);
    G_MapVote_VoteIsFinished = True;
}

main() {
    declare netwrite Text[Text] Net_MapVote_PlayersVotes for Teams[0];
    declare netwrite Boolean Net_MapVote_VoteIsFinished for Teams[0];
    declare netwrite K_MapInfo[] Net_MapVote_MapsToVote for Teams[0];
    
    declare persistent Text[Text] Persistent_MapVote_MapThumbnailUrls;

    ServerAdmin.Authentication_GetToken(NullId, "NadeoLiveServices");
    Layers::Load();
    Layers::Create("MapVote", GetManialink());
    G_MapVote_NextMapUid = MapList[NextMapIndex].MapUid;

    while (True) {
        yield;
        Net_MapVote_VoteIsFinished = G_MapVote_VoteIsFinished;
        Net_MapVote_PlayersVotes = G_MapVote_PlayersVotes;
        foreach(Event in PendingEvents) {
            if (Event == Null) continue;
            
            if (Event.Type == CServerPluginEvent::EType::MapUnloadRequested) {
                StartVote();
            } else if (Event.Type == CServerPluginEvent::EType::ModeCallback) {
                if (Event.ModeCallbackType == "Maniaplanet.StartMap_Start") {
                    if (MapList.existskey(NextMapIndex)) {
                        G_MapVote_NextMapUid = MapList[NextMapIndex].MapUid;
                    }
                }
            }
        }

        if (MapUnloadRequested && G_MapVote_CanVote) {
            
            Layers::Attach("MapVote");
            ---DuringVotingScreen---
            if (Now >= G_MapVote_EndTime && !G_MapVote_VoteIsFinished) {
                EndVote();
            }
            if (Now >= G_MapVote_EndTime+5000) {
                HoldMapUnloadRequest = False;
                G_MapVote_PlayersVotes = [];
                G_MapVote_VoteTimes = [];
                G_MapVote_MapInfoRequests = [];
                UIManager.UIAll.ForceSpectator = False;
            }
        } else {
            Layers::Detach("MapVote"); 
            G_MapVote_VoteIsFinished = False;
        }
    }
}
